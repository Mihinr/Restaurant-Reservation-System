// Transaction with optimistic locking
async function updateReservation(
  reservationId: string,
  updates: Partial<Reservation>,
  expectedVersion: number
): Promise<Reservation> {
  return await prisma.$transaction(async (tx) => {
    // Check version first
    const existing = await tx.reservation.findUnique({
      where: { id: reservationId, version: expectedVersion }
    });
    
    if (!existing) {
      throw new AppError(409, 'Reservation was modified by another user');
    }
    
    // Proceed with update
    return await tx.reservation.update({
      where: { id: reservationId },
      data: { ...updates, version: { increment: 1 } }
    });
  });
}

// store/slices/authSlice.ts
import { createSlice, createAsyncThunk } from '@reduxjs/toolkit';
import { authService } from '../../services/authService';

interface AuthState {
  user: User | null;
  token: string | null;
  isLoading: boolean;
  error: string | null;
}

const initialState: AuthState = {
  user: null,
  token: localStorage.getItem('token'),
  isLoading: false,
  error: null,
};

export const login = createAsyncThunk(
  'auth/login',
  async (credentials: LoginCredentials, { rejectWithValue }) => {
    try {
      return await authService.login(credentials);
    } catch (error) {
      return rejectWithValue(error.message);
    }
  }
);

const authSlice = createSlice({
  name: 'auth',
  initialState,
  reducers: {
    logout: (state) => {
      state.user = null;
      state.token = null;
      localStorage.removeItem('token');
    },
  },
  extraReducers: (builder) => {
    builder
      .addCase(login.pending, (state) => {
        state.isLoading = true;
        state.error = null;
      })
      .addCase(login.fulfilled, (state, action) => {
        state.isLoading = false;
        state.user = action.payload.user;
        state.token = action.payload.accessToken;
        localStorage.setItem('token', action.payload.accessToken);
      })
      .addCase(login.rejected, (state, action) => {
        state.isLoading = false;
        state.error = action.payload as string;
      });
  },
});

export const { logout } = authSlice.actions;
export default authSlice.reducer;

// pages/Reservation/CreateReservation.tsx
const steps = ['Search', 'Select Table', 'Details', 'Confirmation'];

export const CreateReservationPage: React.FC = () => {
  const [currentStep, setCurrentStep] = useState(0);
  const [reservationData, setReservationData] = useState<Partial<Reservation>>({});

  const handleNext = (data: Partial<Reservation>) => {
    setReservationData(prev => ({ ...prev, ...data }));
    setCurrentStep(prev => prev + 1);
  };

  const handleBack = () => {
    setCurrentStep(prev => prev - 1);
  };

  const renderStep = () => {
    switch (currentStep) {
      case 0: return <SearchStep onNext={handleNext} />;
      case 1: return <TableSelectionStep onNext={handleNext} onBack={handleBack} />;
      case 2: return <DetailsStep onNext={handleNext} onBack={handleBack} />;
      case 3: return <ConfirmationStep data={reservationData} onBack={handleBack} />;
      default: return null;
    }
  };

  return (
    <div className="container mx-auto px-4 py-8">
      <Stepper steps={steps} currentStep={currentStep} />
      {renderStep()}
    </div>
  );
};

        partySize: 4,
      };

      const response = await request(app)
        .post('/api/v1/reservations')
        .set('Authorization', `Bearer ${authToken}`)
        .send(reservationData);

      expect(response.status).toBe(201);
      expect(response.body).toHaveProperty('id');

    });

        .post('/api/v1/reservations')
        .set('Authorization', `Bearer ${authToken}`)
